```cpp
{
    return float4(input.color, 1.0f);
}
```

## C++
VertexShader부터 보면 이해하기 조금 헷갈려서 c++코드부터 보자
```cpp
struct NormalVertexConstantBuffer {
    float scale = 0.1f; // 노멀 그릴 선분 스케일
    float dummy[3];
};
```
우선 노멀벡터를 그릴때 그릴 길이를 받아오기 위해 ConstantBuffer에 scale값을 추가한다.
padding 더비 바이트 추가된것만 주의(16바이트씩 전송 기억)

이제 MakeBox처럼 노멀이라는 오브젝트를 그릴차례다
```cpp
// 노멀 벡터 그리기
    // 문제를 단순화하기 위해 InputLayout은 BasicVertexShader와 같이 사용합시다.
    m_normalLines = std::make_shared<Mesh>();

    std::vector<Vertex> normalVertices;
    std::vector<uint16_t> normalIndices;
    for (size_t i = 0; i < meshData.vertices.size(); i++) {

        auto v = meshData.vertices[i];

        v.texcoord.x = 0.0f; // 시작점 표시
        normalVertices.push_back(v);

        v.texcoord.x = 1.0f; // 끝점 표시
        normalVertices.push_back(v);

        normalIndices.push_back(uint16_t(2 * i));
        normalIndices.push_back(uint16_t(2 * i + 1));
    }
```
meshData = Box의 Mesh정보
Box의 Vertex를 모두 순회하면서 Vertex마다 2개의 normal를 지정해준다.
노멀을 그릴때 texcoord가 필요없어서 편의상 트릭으로 여기서 texcoord.x 에 시작점,끝점 인지 표시해준다. 이 표시를 통해 쉐이더에서 두점을 정의하고 그리게됨.
마찬가지로 index도 넣어준다.

Normal을 그리기 위한 필요한 Shader,buffer Create하는 단계
CreateVertexBuffer,CreateIndexBuffer,CreateVertexShaderAndInputLayout,CreatePixelShader
inputlayout은 basicInputElements 이거 기존꺼 그대로 재사용
```cpp
// Create Vertex,Index Buffer for noraml
    AppBase::CreateVertexBuffer(normalVertices, m_normalLines->m_vertexBuffer);
    m_normalLines->m_indexCount = UINT(normalIndices.size());
    AppBase::CreateIndexBuffer(normalIndices, m_normalLines->m_indexBuffer);
    // Create ConstantBuffer for normal
    AppBase::CreateConstantBuffer(m_normalVertexConstantBufferData,
                                  m_normalLines->m_vertexConstantBuffer);
    AppBase::CreateVertexShaderAndInputLayout(
        L"NormalVertexShader.hlsl", basicInputElements, m_normalVertexShader,
        m_basicInputLayout);

    AppBase::CreatePixelShader(L"NormalPixelShader.hlsl", m_normalPixelShader);
```

Render
Render하기전에 normal용으 생성한 VertexShader와 PixelShader를 설정해주고 
vertexbuffer와constantbuffer,Indexbuffer도 세팅해준다.
```cpp

    if (m_drawNormals) {

        // TODO: 여기에 필요한 내용들 작성
        m_context->PSSetShader(m_normalPixelShader.Get(), 0, 0);

        m_context->VSSetShader(m_normalVertexShader.Get(), 0, 0);

        ID3D11Buffer *pptr[2] = {m_mesh->m_vertexConstantBuffer.Get(),
                                 m_normalLines->m_vertexConstantBuffer.Get()};
        m_context->VSSetConstantBuffers(0, 2, pptr);

        m_context->IASetVertexBuffers(
            0, 1, m_normalLines->m_vertexBuffer.GetAddressOf(), &stride,
            &offset);
        m_context->IASetIndexBuffer(m_normalLines->m_indexBuffer.Get(),
                                    DXGI_FORMAT_R16_UINT, 0);
        m_context->IASetPrimitiveTopology(D3D11_PRIMITIVE_TOPOLOGY_LINELIST);
        m_context->DrawIndexed(m_normalLines->m_indexCount, 0, 0);
    }
```
IASetPrimitiveTopology(D3D11_PRIMITIVE_TOPOLOGY_LINELIST);
여기서 중요한점은  primiveTopology를 D3D11_PRIMITIVE_TOPOLOGY_LINELIST으로 설정한다는 점이다. 그래야 2개씩 끊어서 선분으로 그릴 수 있음

**중요**
VSSetConstantBuffers 이 부분도 중요한데 normalShader에서도 변환행렬이 필요하다.
근데 이미 box의 constantbuffer에 해당 데이터가 다 들어있다. 따라서 따로 normalconstant버퍼에 또 만들어 GPU에 게속 update할 필요 없이 VSSetConstantBuffers()에 이미 GPU에 올라있는 m_mesh의 constant버퍼와 normal의 constantbuffer의 포인터를 넘겨주면 쉐이더에서 사용가능하다.


## NormalVertexShader
```cpp
#include "Common.hlsli"
cbuffer BasicVertexConstantBuffer : register(b0)
{
    matrix model;
    matrix invTranspose;
    matrix view;
    matrix projection;
};

cbuffer NormalVertexConstantBuffer : register(b1)
{
    //matrix model;
    //matrix invTranspose;
    //matrix view;
    //matrix projection;
    float scale; // 그려지는 선분의 길이 조절
};
```
앞서 box의constantbuffer정보와 normal의 constantbuffer정보를 같이 전달 했기에 둘다 받는다. register(b0), register(b1) 전달된 순서대로 해줘야함
NormalVertexConstantBuffer에는 선분의 길이에 해당할 scale만 추가가 되었다.

main
```cpp
PixelShaderInput main(VertexShaderInput input)
{
    PixelShaderInput output;
    float4 pos = float4(input.posModel, 1.0f);

    // Normal 먼저 변환
    float4 normal = float4(input.normalModel, 0.0f);
    output.normalWorld = mul(normal, invTranspose).xyz;
    output.normalWorld = normalize(output.normalWorld);
    
    pos = mul(pos, model);
    
    float t = input.texcoord.x;
    
    pos.xyz += output.normalWorld * t * scale;

    output.posWorld = pos.xyz;
    
    pos = mul(pos, view);
    pos = mul(pos, projection);

    output.posProj = pos;
    output.texcoord = input.texcoord;
    
    output.color = float3(1.0, 1.0, 0.0) * (1.0 - t) + float3(1.0, 0.0, 0.0) * t;

    return output;
}
```
우선 한 vertex에 대하여 2번씩 현재 쉐이더로 들어올것이다. 
왜냐하면 우리가 c++코드에서 앞서 노멀을 만들 때 ,한 Vertex에 대하여 2개의 점을 만드는데 하나는 시작점 하나는 끝점으로 texcoord.x에 플래그를 넣어 표시해줬기 때문이다.

현재 vertex의 노멀을 가져와 invTranspose을 곱해 온전한 노멀을 구한 후
현재 vertex에 model행렬을 곱해 변환된 vertex값을 가져온다.
texcoord를 t에 저장해주고 있는데 우리가 지정한 플래그인 0(시작점) 아니면 1(끝점)이다.

pos.xyz += output.normalWorld * t * scale;
이 부분 같은 경우 시작점이면 t가 0이라 원래 pos.xyz값이되고
끝점이면 t가 1이라 원래 pos.xyz에서 노멀벡터 방향으로 스케일만큼 더해진 위치가 될 것이다.

그 다음 노멀벡터 선분을 그릴때도 래스터화가 필요하므로 아래 과정을 마저 진행
output.posWorld = pos.xyz;

pos = mul(pos, view);
pos = mul(pos, projection);

output.posProj = pos;
output.texcoord = input.texcoord;

컬러 값도 t 값으로 시작점 끝점의 색깔을 다르게한다.
