앞서 만든 그리드 만드는 원리를 토대로 실린더를 만들어보겠다.
그리드 평면을 둥글게 만단고 생각하면 된다.

```cpp
MeshData GeometryGenerator::MakeCylinder(const float bottomRadius,
                                         const float topRadius, float height,
                                         int sliceCount) {

    using namespace DirectX;
    using DirectX::SimpleMath::Matrix;
    using DirectX::SimpleMath::Vector3;

    // Texture 좌표계때문에 (sliceCount + 1) x 2 개의 버텍스 사용

    const float dTheta = -XM_2PI / float(sliceCount);

    MeshData meshData;

    vector<Vertex> &vertices = meshData.vertices;

   
    Vector3 StartPoint(0.0f, height * -0.5f, bottomRadius * -0.5f);
    Vector3 StartNormal(0.0f, 0.0f, -1.0f);
    // 옆면의 바닥 버텍스들 (인덱스 0 이상 sliceCount 미만)
    for (int i = 0; i <= sliceCount; i++) {
        Vertex v;

        v.position = Vector3::Transform(StartPoint, Matrix::CreateRotationY(dTheta * i));
        v.normal = Vector3::Transform(StartNormal, Matrix::CreateRotationY(dTheta * i));
        v.normal.Normalize();
        v.texcoord = Vector2( float(i) / sliceCount, 1.0f);
        vertices.push_back(v);
    }

    StartPoint = Vector3(0.0f, height * 0.5f, topRadius * -0.5f);
    // 옆면의 맨 위 버텍스들 (인덱스 sliceCount 이상 2 * sliceCount 미만)
    for (int i = 0; i <= sliceCount; i++) {
        Vertex v;
        
        v.position = Vector3::Transform(StartPoint, Matrix::CreateRotationY(dTheta * i));
        v.normal = Vector3::Transform(StartNormal, Matrix::CreateRotationY(dTheta * i));
        v.normal.Normalize();
        v.texcoord = Vector2( float(i) / sliceCount, 0.0f);
        vertices.push_back(v);
    }

    vector<uint16_t> &indices = meshData.indices;

    for (int i = 0; i <= sliceCount; i++) {
        // TODO: 삼각형 두 개 씩
        indices.push_back(i);
        indices.push_back(i + sliceCount + 1);
        indices.push_back(i + 1);
       
        indices.push_back(i);
        indices.push_back(i + sliceCount);
        indices.push_back(i + sliceCount + 1);
    }

    return meshData;
}
```
