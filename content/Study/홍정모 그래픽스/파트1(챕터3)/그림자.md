참고로 이번 방법은 그림자에 가까운 효과를 나타내는것이지 실제로 리얼타임 그래픽스에서는 이렇게 구현하지 않는다.

구현원리
여기서 부턴 광원이라는 개념이 등장.
눈으로부터 물체에 레이를 쏘고 만나는 지점에서 광원으로 다시 레이를 쐈을때 가리는 물체가 있다면 그림자 효과를 적용. 여기서는 ambient컬러만 적용하는것으로 그림자처럼 보이게 할것이다.
![[Pasted image 20240809224254.png]]
```cpp
// 광선이 물체에 닿으면 그 물체의 색 반환
		vec3 traceRay(Ray &ray)
		{
			// Render first hit
			const auto hit = FindClosestCollision(ray);

			if (hit.d >= 0.0f)
			{
				vec3 pointToDir = glm::normalize(light.pos - hit.point);
				vec3 shadowRayPoint = hit.point + pointToDir * 1e-4f;

				Ray shadowRay;
				shadowRay.dir = pointToDir;
				shadowRay.start = shadowRayPoint;

				// 물체에 가려졌다면  amb값만 계산하여 그림자 효과 적용
				const auto shadowRayHit = FindClosestCollision(shadowRay);
				if (shadowRayHit.d >= 0.0f)
				{
					return hit.obj->amb;
				}

				// Diffuse
				const vec3 dirToLight = glm::normalize(light.pos - hit.point);
				const float diff = glm::max(dot(hit.normal, dirToLight), 0.0f);

				// Specular
				const vec3 reflectDir = 2.0f * dot(hit.normal, dirToLight) * hit.normal - dirToLight;
				const float specular = glm::pow(glm::max(glm::dot(-ray.dir, reflectDir), 0.0f), hit.obj->alpha);

				return hit.obj->amb + hit.obj->dif * diff + hit.obj->spec * specular;
			}

			return vec3(0.0f);
		}
```
