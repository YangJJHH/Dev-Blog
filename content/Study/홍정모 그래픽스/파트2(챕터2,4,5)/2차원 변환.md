회전 코드
```cpp
vec3 RotateAboutZ(const vec3 &v, const float &theta) {

    vec3 rotatePos(0.0f, 0.0f, v.z);
    rotatePos.x = v.x * cos(theta) - v.y * sin(theta);
    rotatePos.y = v.x * sin(theta) + v.y * cos(theta);
    return rotatePos;
}
```


Rasterization::Update() 코드
결국 아래의 코드처럼 좌표변환이 VertexShader가 하는 일들이다.
```cpp
 // GUI 입력에 대해서 vertexBuffer 업데이트
    // 이 예제에서는 this->vertexBuffer만 업데이트하고
    // colorBuffer와 indexBuffer는 변화 없음

    for (size_t i = 0; i < circle.vertices.size(); i++) {

        // 회전 1
        auto temp = RotateAboutZ(circle.vertices[i], this->rotation1);
        // 스케일
        temp = temp * vec3(this->scaleX, this->scaleY, 1.0f);
        // 이동 1
        temp = temp + this->translation1;
        // 회전 2
        temp = RotateAboutZ(temp, this->rotation2);
        // 이동 2
        temp = temp + +this->translation2;

        // 변환된 좌표를 반영
        this->vertexBuffer[i] = temp;
    }
```


전체 코드
우선 전체적인 흐름을 설명하자면 vertexbuffer에 그릴 오브젝트의 vertex를 저장하고 DrawIndexedTraingle를 호출하여 해당 오브젝트의 vertex를 랜더링 하는 것이다.
그래서 vertexbuffer를 오브젝트 상황에 맞게 변형하고 DrawIndexedTraingle를 호출하여 그리고를 반복하는 것을 볼 수 있다.


```cpp
void Rasterization::Render(vector<vec4> &pixels) {

    // 그래픽카드(GPU)의 특징
    // 1. GPU가 훨씬 빠릅니다. 가급적 GPU 사용
    // 2. CPU->GPU 복사가 느립니다. 한 번 넣어두고 반복 사용
    // 3. GPU->CPU 복사는 더 느립니다.

    // 태양 그리기
    this->vertexBuffer.resize(sun.vertexBuffer.size());
    for (size_t i = 0; i < sun.vertexBuffer.size(); i++) {
        this->vertexBuffer[i] = sun.vertexBuffer[i];
    }

    this->indexBuffer = sun.indexBuffer;
    this->colorBuffer = sun.colorBuffer;

    // 현재 버퍼로 삼각형 하나씩 그리기 (아래 for루프는 여러번 사용됨)
    for (size_t i = 0; i < this->indexBuffer.size(); i += 3) {
        DrawIndexedTriangle(i, pixels);
    }

    // 지구 그리기
    this->vertexBuffer.resize(earth.vertexBuffer.size());
    for (size_t i = 0; i < earth.vertexBuffer.size(); i++) {
        // 우선 태양으로부터 거리만큼 이동
        auto temp = earth.vertexBuffer[i] + vec3(distSunToEarth, 0.0f, 0.0f);
        // 현재 업데이트된 공전 각도 만큼 회전
        temp = RotateAboutZ(temp, earthAngle);
        // 계산된 버텍스 버퍼에 반영
        this->vertexBuffer[i] = temp;
    }

    this->indexBuffer = earth.indexBuffer;
    this->colorBuffer = earth.colorBuffer;

    for (size_t i = 0; i < this->indexBuffer.size(); i += 3) {
        DrawIndexedTriangle(i, pixels);
    }

    // 달 그리기
    this->vertexBuffer.resize(moon.vertices.size());
    for (size_t i = 0; i < moon.vertices.size(); i++) {
        // 달 공전 거리 만큼 이동
        auto temp = moon.vertexBuffer[i] + vec3(distEarthToMoon, 0.0f, 0.0f);
        // 원점을 기준으로 회전
        temp = RotateAboutZ(temp, moonAngle);
        // 달의 회전중심 좌표를 원점에서 지국 중심으로 이동
        float earthRadius = 0.05f;
        vec3 EarthCenterPos =  vec3(cos(earthAngle), sin(earthAngle), 0.0f) * distSunToEarth;
        temp = temp + EarthCenterPos;
        this->vertexBuffer[i] = temp;
    }

    this->indexBuffer = moon.indexBuffer;
    this->colorBuffer = moon.colorBuffer;

    for (size_t i = 0; i < this->indexBuffer.size(); i += 3) {
        DrawIndexedTriangle(i, pixels);
    }
}
```

태양은 움직이지 않기 때문에 기존 Vertex를 그대로 vertexbuffer에 복사하여 랜더.

그다음 vertexbuffer를 초기화하고 지구에 대한 vertex로 채운다
지구는 태양을 중심으로 회전하기 때문에 먼저 공전 거리만큼 이동 시켜주고 원점을 중심으로 각속도만큼 회전한 각에 대해 회전을 적용하여 공전을 구현하였다

달은 지구를 기준으로 공전하는데, 
1. 우선 달과 지구 거리만큼 원점에서 이동시켜준다.
2. 그 다음 해당 위치에서 원점을 기준으로 회전시킨다. 여기까지는 공전 거리는 같지만 중심이 지구가 아니라 원점임
3. 지구를 중심으로 하기 위해 현재 지구 중심좌표를 구하여 해당 위치로 중심을 이동시킨다.
이와 같은 이동-회전-이동으로 달의 공전을 구현하였다.
